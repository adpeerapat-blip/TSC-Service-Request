function doGet(e) {
    if (e.postData && e.postData.contents) {
        const postParams = JSON.parse(e.postData.contents);
        e.parameter = { ...e.parameter, ...postParams };
         return handleRequest(e);
    }
    return HtmlService.createHtmlOutput("Google Apps Script is running.");
}


function doPost(e) {
    const params = JSON.parse(e.postData.contents);
    e.parameter = params;
    return handleRequest(e);
}


function handleRequest(e) {
    try {
        const action = e.parameter.action;

        switch (action) {
            case 'login': return loginUser(e.parameter.email, e.parameter.password);
            case 'signup': return signupUser(e.parameter);
            case 'forgotPassword': return forgotPassword(e.parameter.email);
            case 'getUserData': return getUserData(e.parameter.email);
            case 'updateProfile': return updateUserProfile(e.parameter);
            case 'updateUserSignature': return updateUserSignature(e.parameter);
            case 'createNewJob': return processNewJob(e.parameter);
            case 'getAllJobs':
                const userEmailForJobs = e.parameter.userEmail;
                if (!userEmailForJobs) throw new Error("ต้องระบุ userEmail เพื่อดึงข้อมูลใบงาน");
                const usersMap = readFromFirebase("users") || {};
                const encodedEmail = encodeEmail(userEmailForJobs);
                const user = usersMap[encodedEmail];
                if (user) {
                    const role = user.Role;
                    const department = user.Department;
                    return getAllJobs(role, department);
                }
                return getAllJobs("UNKNOWN_USER");
            case 'getJobDetails': return getJobDetails(e.parameter);
            case 'processApproval': return processApproval(e.parameter);
            case 'acceptJob': return acceptJob(e.parameter);
            case 'declineJob': return declineJob(e.parameter);
            case 'closeJob': return closeJob(e.parameter);
            case 'updateJobStatus': return updateJobStatus(e.parameter);
            case 'getDropdownData': return getDropdownData();
            case 'addDropdownItem': return addDropdownItem(e.parameter.type, e.parameter.value);
            case 'updateDropdownItem': return updateDropdownItem(e.parameter.type, e.parameter.oldValue, e.parameter.newValue);
            case 'deleteDropdownItem': return deleteDropdownItem(e.parameter.type, e.parameter.value);
            case 'getAllUsers': return getAllUsers();
            case 'updateUserRole': return updateUserRole(e.parameter);
            case 'updateUserDepartment': return updateUserDepartment(e.parameter);
            case 'deleteUser': return deleteUser(e.parameter);
            case 'updateUserApprovalLevel': return updateUserApprovalLevel(e.parameter);
            case 'updateUserCanAcceptJobs': return updateUserCanAcceptJobs(e.parameter);
            // Customer Actions
            case 'getAllCustomers': return getAllCustomers();
            case 'addOrUpdateCustomer': return addOrUpdateCustomer(e.parameter);
            case 'deleteCustomer': return deleteCustomer(e.parameter);

            default: throw new Error("Action ที่ระบุไม่ถูกต้อง");
        }
    } catch (error) {
        Logger.log(`Request Error: ${error.toString()}\nStack: ${error.stack}`);
        return ContentService
            .createTextOutput(JSON.stringify({ "status": "error", "message": error.message }))
            .setMimeType(ContentService.MimeType.JSON);
    }
}

// --- Firebase Core Utilities ---

function getFirebaseUrl(path) {
  let url = FIREBASE_DB_URL;
  if (!url.endsWith('/')) {
    url += '/';
  }
  url += path + '.json';
  if (FIREBASE_AUTH_SECRET) {
    url += '?auth=' + FIREBASE_AUTH_SECRET;
  }
  return url;
}

function encodeEmail(email) {
  if (!email) return "";
  return email.replace(/\./g, ',');
}

function decodeEmail(encodedEmail) {
  if (!encodedEmail) return "";
  return encodedEmail.replace(/,/g, '.');
}

function readFromFirebase(path) {
  const url = getFirebaseUrl(path);
  const res = UrlFetchApp.fetch(url, { method: "get", muteHttpExceptions: true });
  if (res.getResponseCode() === 200) {
    const text = res.getContentText();
    return text ? JSON.parse(text) : null;
  }
  Logger.log("Firebase GET failed for path " + path + ": " + res.getContentText());
  return null;
}

function writeToFirebase(path, data) {
  const url = getFirebaseUrl(path);
  const options = {
    method: "put",
    contentType: "application/json",
    payload: JSON.stringify(data),
    muteHttpExceptions: true
  };
  const res = UrlFetchApp.fetch(url, options);
  if (res.getResponseCode() === 200) {
    return true;
  }
  Logger.log("Firebase PUT failed for path " + path + ": " + res.getContentText());
  return false;
}

function patchFirebase(path, data) {
  const url = getFirebaseUrl(path);
  const options = {
    method: "patch",
    contentType: "application/json",
    payload: JSON.stringify(data),
    muteHttpExceptions: true
  };
  const res = UrlFetchApp.fetch(url, options);
  if (res.getResponseCode() === 200) {
    return true;
  }
  Logger.log("Firebase PATCH failed for path " + path + ": " + res.getContentText());
  return false;
}

// --- Authentication Functions ---

function loginUser(email, password) {
    if (!email || !password) throw new Error("ต้องระบุอีเมลและรหัสผ่าน");
    const encodedEmail = encodeEmail(email);
    const user = readFromFirebase("users/" + encodedEmail);
    if (user) {
        const storedHash = user.PasswordHash;
        const inputHash = hashPassword(password);
        if (storedHash === inputHash) {
            return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "เข้าสู่ระบบสำเร็จ" })).setMimeType(ContentService.MimeType.JSON);
        }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "อีเมลหรือรหัสผ่านไม่ถูกต้อง" })).setMimeType(ContentService.MimeType.JSON);
}

function signupUser(params) {
    const { fullName, email, phone, department, password } = params;
    if (!fullName || !email || !phone || !department || !password) throw new Error("ต้องกรอกข้อมูลทุกช่องเพื่อสมัครสมาชิก");
    
    const encodedEmail = encodeEmail(email);
    const existingUser = readFromFirebase("users/" + encodedEmail);
    if (existingUser) {
        throw new Error("อีเมลนี้ถูกลงทะเบียนแล้ว");
    }
    
    addDropdownItem('department', department, true);
    const hash = hashPassword(password);
    
    const rawUsers = readFromFirebase("users") || {};
    const newUserId = `USER-${(Object.keys(rawUsers).length + 1).toString().padStart(3, '0')}`;
    const newUser = {
        UserID: newUserId,
        FullName: fullName,
        Email: email,
        Phone: String(phone),
        Department: department,
        PasswordHash: hash,
        SignatureURL: '',
        Role: 'User',
        ApprovalLevel: '',
        CanAcceptJobs: 'FALSE'
    };
    writeToFirebase("users/" + encodedEmail, newUser);

    return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "สมัครสมาชิกสำเร็จ! กรุณาเข้าสู่ระบบ" })).setMimeType(ContentService.MimeType.JSON);
}

function forgotPassword(email) {
    if (!email) throw new Error("ต้องระบุอีเมล");
    const encodedEmail = encodeEmail(email);
    const user = readFromFirebase("users/" + encodedEmail);
    if (!user) throw new Error("ไม่พบอีเมลนี้ในระบบ");

    const newPassword = Math.random().toString(36).slice(-8);
    const hash = hashPassword(newPassword);
    
    patchFirebase("users/" + encodedEmail, { PasswordHash: hash });

    MailApp.sendEmail(email, "รหัสผ่านใหม่สำหรับ WebApp ศูนย์รวมช่าง", `รหัสผ่านใหม่ของคุณคือ: ${newPassword}`);
    return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "รหัสผ่านใหม่ถูกส่งไปยังอีเมลของคุณแล้ว" })).setMimeType(ContentService.MimeType.JSON);
}

function getUserData(userEmail) {
    if (!userEmail) throw new Error("ต้องระบุพารามิเตอร์อีเมล");
    const encodedEmail = encodeEmail(userEmail);
    const user = readFromFirebase("users/" + encodedEmail);
    if (user) {
        const userData = { ...user };
        delete userData.PasswordHash;
        return ContentService.createTextOutput(JSON.stringify({ status: "success", data: userData })).setMimeType(ContentService.MimeType.JSON);
    }
    throw new Error("ไม่พบผู้ใช้งาน");
}

function getDropdownData() {
    const rawDropdownsMap = readFromFirebase("dropdowns") || {};
    const rawDropdowns = Object.values(rawDropdownsMap);
    const dropdowns = { jobTypes: [], customerTypes: [], departments: [] };
    rawDropdowns.forEach(row => {
        if (row.JobType) dropdowns.jobTypes.push(row.JobType);
        if (row.CustomerType) dropdowns.customerTypes.push(row.CustomerType);
        if (row.Department) dropdowns.departments.push(row.Department);
    });
    return ContentService.createTextOutput(JSON.stringify({ status: "success", data: dropdowns })).setMimeType(ContentService.MimeType.JSON);
}

// --- Dropdown CRUD Functions ---

function addDropdownItem(type, value, silent = false) {
    let rawDropdownsMap = readFromFirebase("dropdowns") || {};
    const rawDropdowns = Object.values(rawDropdownsMap);
    
    const headerMap = {"jobType": "JobType", "customerType": "CustomerType", "department": "Department"};
    const columnName = headerMap[type];
    if (!columnName) throw new Error("ประเภท dropdown ไม่ถูกต้อง");

    const columnValues = rawDropdowns.map(row => row[columnName]).filter(Boolean);
    if (columnValues.includes(value)) {
        if (silent) return; 
        throw new Error("มีข้อมูลนี้อยู่แล้ว");
    }
    
    // Find empty slot in Map or write a new node
    const keys = Object.keys(rawDropdownsMap);
    let addedKey = null;
    for (let i = 0; i < keys.length; i++) {
        const key = keys[i];
        if (!rawDropdownsMap[key][columnName]) {
            addedKey = key;
            break;
        }
    }
    
    if (addedKey) {
        rawDropdownsMap[addedKey][columnName] = value;
        writeToFirebase("dropdowns/" + addedKey, rawDropdownsMap[addedKey]);
    } else {
        const newKey = "item_" + keys.length;
        const newRow = {};
        newRow[columnName] = value;
        writeToFirebase("dropdowns/" + newKey, newRow);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "เพิ่มข้อมูลสำเร็จ" })).setMimeType(ContentService.MimeType.JSON);
}

function updateDropdownItem(type, oldValue, newValue) {
    let rawDropdownsMap = readFromFirebase("dropdowns") || {};
    const headerMap = {"jobType": "JobType", "customerType": "CustomerType", "department": "Department"};
    const columnName = headerMap[type];
    if (!columnName) throw new Error("ประเภท dropdown ไม่ถูกต้อง");
    
    let updated = false;
    for (const key in rawDropdownsMap) {
        if (rawDropdownsMap[key][columnName] === oldValue) {
            rawDropdownsMap[key][columnName] = newValue;
            writeToFirebase("dropdowns/" + key, rawDropdownsMap[key]);
            updated = true;
        }
    }
    
    if (updated) {
        return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "อัปเดตข้อมูลสำเร็จ" })).setMimeType(ContentService.MimeType.JSON);
    }
    throw new Error("ไม่พบข้อมูล");
}

function deleteDropdownItem(type, value) {
    let rawDropdownsMap = readFromFirebase("dropdowns") || {};
    const headerMap = {"jobType": "JobType", "customerType": "CustomerType", "department": "Department"};
    const columnName = headerMap[type];
    if (!columnName) throw new Error("ประเภท dropdown ไม่ถูกต้อง");
    
    let updated = false;
    for (const key in rawDropdownsMap) {
        if (rawDropdownsMap[key][columnName] === value) {
            delete rawDropdownsMap[key][columnName];
            if (Object.keys(rawDropdownsMap[key]).length === 0) {
              UrlFetchApp.fetch(getFirebaseUrl("dropdowns/" + key), { method: "delete" });
            } else {
              writeToFirebase("dropdowns/" + key, rawDropdownsMap[key]);
            }
            updated = true;
        }
    }
    
    if (updated) {
        return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "ลบข้อมูลสำเร็จ" })).setMimeType(ContentService.MimeType.JSON);
    }
    throw new Error("ไม่พบข้อมูล");
}

function getAllJobs(role, department) {
    const allJobsMap = readFromFirebase("jobs") || {};
    const allJobs = Object.values(allJobsMap);
    let jobsData = (role === 'User' && department) ? allJobs.filter(row => row && row["หน่วยงาน"] === department) : allJobs;
    if (role === 'UNKNOWN_USER') jobsData = [];
    jobsData = jobsData.filter(Boolean);
    return ContentService.createTextOutput(JSON.stringify({ status: "success", data: jobsData })).setMimeType(ContentService.MimeType.JSON);
}

function getJobDetails(params) {
    const { jobId } = params;
    if (!jobId) throw new Error("ต้องระบุ Job ID");

    const jobDetails = readFromFirebase("jobs/" + jobId);
    if (!jobDetails) throw new Error("ไม่พบใบงาน");

    const usersMap = readFromFirebase("users") || {};
    const users = Object.values(usersMap);
    const userMap = users.reduce((map, u) => {
        if (u && u.Email) map[u.Email] = { name: u.FullName, signature: u.SignatureURL };
        return map;
    }, {});

    const approverL1 = userMap[jobDetails["ผู้อนุมัติ Level 1"]] || {};
    jobDetails.SignatureL1 = approverL1.signature || '';
    jobDetails.ApproverL1Name = approverL1.name || '...................................';
    
    const approverL2 = userMap[jobDetails["ผู้อนุมัติ Level 2"]] || {};
    jobDetails.SignatureL2 = approverL2.signature || '';
    jobDetails.ApproverL2Name = approverL2.name || '...................................';

    const actualReceiver = userMap[jobDetails["ผู้รับงานจริง"]] || {};
    jobDetails.ActualReceiverName = actualReceiver.name || jobDetails['ผู้รับผิดชอบ'] || '...................................';
    jobDetails.ActualReceiverSignature = actualReceiver.signature || '';

    return ContentService.createTextOutput(JSON.stringify({ status: "success", data: jobDetails })).setMimeType(ContentService.MimeType.JSON);
}

function processApproval(params) {
    const { jobId, isApproved, reason, approverEmail, assignedTo } = params;
    const jobData = readFromFirebase("jobs/" + jobId);
    if (!jobData) throw new Error("ไม่พบใบงาน");
    const currentStatus = jobData["สถานะ"];
    const updateData = {};

    if (currentStatus === "รออนุมัติ Level 1") {
        updateData["ผู้อนุมัติ Level 1"] = approverEmail;
        updateData["เหตุผล L1"] = reason;
        updateData["วันที่ L1"] = new Date().toISOString();
        if (isApproved) {
            updateData["ผู้รับผิดชอบ"] = assignedTo.join(',');
            updateData["สถานะ"] = "รออนุมัติ Level 2";
        } else {
            updateData["สถานะ"] = "ไม่อนุมัติ L1";
        }
    } else if (currentStatus === "รออนุมัติ Level 2") {
        updateData["ผู้อนุมัติ Level 2"] = approverEmail;
        updateData["เหตุผล L2"] = reason;
        updateData["วันที่ L2"] = new Date().toISOString();
        updateData["สถานะ"] = isApproved ? "รอรับงาน" : "ไม่อนุมัติ L2";
    } else throw new Error("ใบงานนี้ไม่อยู่ในสถานะที่จะอนุมัติได้");
    
    patchFirebase("jobs/" + jobId, updateData);
    return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "ดำเนินการอนุมัติสำเร็จ" })).setMimeType(ContentService.MimeType.JSON);
}

function declineJob(params) {
    const { jobId, technicianEmail, reason } = params;
    const jobData = readFromFirebase("jobs/" + jobId);
    if (!jobData) throw new Error("ไม่พบใบงาน");
    
    const assignedCol = jobData["ผู้รับผิดชอบ"] || '';
    const assignedTechnicians = assignedCol.split(',');
    
    let declinedList = jobData["ผู้ปฏิเสธงาน (JSON)"] ? JSON.parse(jobData["ผู้ปฏิเสธงาน (JSON)"]) : [];
    if (!declinedList.some(d => d.email === technicianEmail)) {
        declinedList.push({ email: technicianEmail, reason });

// --- Helper Functions ---

function deleteFileByUrl(url) {
    if (!url) return;
    try {
        let fileId = '';
        if (url.indexOf('lh3.googleusercontent.com/d/') !== -1) {
            fileId = url.split('lh3.googleusercontent.com/d/')[1].split('/')[0].split('?')[0];
        } else if (url.indexOf('id=') !== -1) {
            fileId = url.split('id=')[1].split('&')[0];
        } else if (url.indexOf('/file/d/') !== -1) {
            fileId = url.split('/file/d/')[1].split('/')[0];
        }
        
        if (fileId) {
            const file = DriveApp.getFileById(fileId);
            file.setTrashed(true);
            Logger.log(`Trashed file: ${fileId}`);
        }
    } catch (e) {
        Logger.log(`Failed to delete file from URL ${url}: ${e.toString()}`);
    }
}

function hashPassword(password) {
    const hash = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, password, Utilities.Charset.UTF_8);
    return hash.map(byte => (byte & 0xFF).toString(16).padStart(2, '0')).join('');
}

function uploadFileToFolder(fileData, folder) {
    try {
        const decodedFile = Utilities.base64Decode(fileData.data);
        const fileBlob = Utilities.newBlob(decodedFile, fileData.mimeType, fileData.fileName);
        const file = folder.createFile(fileBlob);
        return `https://lh3.googleusercontent.com/d/${file.getId()}`;
    } catch (e) {
        Logger.log(`File upload failed: ${e.toString()}`);
        return '';
    }
}

function uploadSignature(signatureDataString, folderId, baseFileName) {
    if (!signatureDataString || !signatureDataString.startsWith('data:image/png;base64,')) return '';
    try {
        const signatureData = signatureDataString.split(",")[1];
        const decodedSignature = Utilities.base64Decode(signatureData);
        const signatureBlob = Utilities.newBlob(decodedSignature, 'image/png', `${baseFileName}.png`);
        const file = DriveApp.getFolderById(folderId).createFile(signatureBlob);
        return `https://lh3.googleusercontent.com/d/${file.getId()}`;
    } catch (e) {
        Logger.log(`Signature upload failed: ${e.toString()}`);
        return '';
    }
}

function createDocumentWithAttachments(spreadsheetName, jobID, params, signatureUrl, attachmentUrls) {

function generateNewJobIDFirebase(allJobs, extraOffset = 0) {
    const now = new Date();
    const prefix = `TSC-JOB${now.getFullYear()}${(now.getMonth() + 1).toString().padStart(2, '0')}-`;
    
    if (!allJobs || allJobs.length === 0) {
        return prefix + (1 + extraOffset).toString().padStart(3, '0');
    }
    
    const jobIDValues = allJobs.map(j => j.JobID).filter(Boolean);
    const currentMonthSequences = jobIDValues
        .filter(id => id && typeof id === 'string' && id.startsWith(prefix))
        .map(id => parseInt(id.split('-')[2], 10))
        .filter(num => !isNaN(num));

    const lastSequence = currentMonthSequences.length > 0 ? Math.max(...currentMonthSequences) : 0;
    const newSequence = lastSequence + 1 + extraOffset;
    
    return prefix + newSequence.toString().padStart(3, '0');
}

function generateNewJobID(sheet) {
    const lock = LockService.getScriptLock();
    lock.waitLock(30000); 
    let newJobID = '';
    try {
        const now = new Date();
        const prefix = `TSC-JOB${now.getFullYear()}${(now.getMonth() + 1).toString().padStart(2, '0')}-`;
        
        const lastRow = sheet.getLastRow();
        if (lastRow < 2) {
            newJobID = prefix + '001';
            return newJobID;
        }
        
        const jobIDColumnValues = sheet.getRange(2, 3, lastRow - 1, 1).getValues().flat();
        
        const currentMonthSequences = jobIDColumnValues
            .filter(id => id && typeof id === 'string' && id.startsWith(prefix))
            .map(id => parseInt(id.split('-')[2], 10))
            .filter(num => !isNaN(num));

        const lastSequence = currentMonthSequences.length > 0 ? Math.max(...currentMonthSequences) : 0;
        const newSequence = lastSequence + 1;
        
        newJobID = prefix + newSequence.toString().padStart(3, '0');

    } finally {
        lock.releaseLock();
    }
    return newJobID;
}

}

function cleanDuplicatePendingJobs() {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sheet = ss.getSheetByName(JOB_DATA_SHEET_NAME);
  if (!sheet) return "ไม่พบชีต DATA ใบงาน";
  
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const jobIDCol = headers.indexOf("JobID");
  const statusCol = headers.indexOf("สถานะ");
  
  if (jobIDCol === -1 || statusCol === -1) return "ไม่พบคอลัมน์ JobID หรือ สถานะ";

  const jobsMap = {};
  for (let i = 1; i < data.length; i++) {
    const jobID = data[i][jobIDCol];
    if (jobID) {
      if (!jobsMap[jobID]) jobsMap[jobID] = [];
      jobsMap[jobID].push({ rowIndex: i + 1, status: data[i][statusCol], timestamp: data[i][0] });
    }
  }

  const rowsToDelete = [];
  Object.keys(jobsMap).forEach(jobID => {
    const group = jobsMap[jobID];
    if (group.length > 1) {
      const hasProcessed = group.some(item => item.status !== "รออนุมัติ Level 1");
      if (hasProcessed) {
        group.forEach(item => { if (item.status === "รออนุมัติ Level 1") rowsToDelete.push(item.rowIndex); });
      } else {
        group.slice(1).forEach(item => { rowsToDelete.push(item.rowIndex); });
      }
    }
  });

  rowsToDelete.sort((a, b) => b - a);
  rowsToDelete.forEach(rowIndex => sheet.deleteRow(rowIndex));
  return "เคลียร์ใบงานที่ซ้ำซ้อนสำเร็จแล้ว! ลบข้อมูลไปทั้งหมด " + rowsToDelete.length + " รายการ";
}

function uploadDataToFirebase() {
  const ui = SpreadsheetApp.getUi();
  const response = ui.alert('ยืนยันการอัพโหลด', 'คุณต้องการอัพโหลดข้อมูลจาก Google Sheets ไปยัง Firebase ใช่หรือไม่? (ข้อมูลเดิมบน Firebase ทั้งหมดจะถูกเขียนทับ)', ui.ButtonSet.YES_NO);
  if (response !== ui.Button.YES) return;

  try {
    const jobs = getSheetDataAsJson(JOB_DATA_SHEET_NAME);
    const jobsMap = {};
    jobs.forEach(j => { if (j.JobID) jobsMap[j.JobID] = j; });

    const users = getSheetDataAsJson(USER_DATA_SHEET_NAME);
    const usersMap = {};
    users.forEach(u => { if (u.Email) usersMap[encodeEmail(u.Email)] = u; });

    const customers = getSheetDataAsJson(CUSTOMER_DATA_SHEET_NAME);
    const customersMap = {};
    customers.forEach(c => { if (c.CustomerID) customersMap[c.CustomerID] = c; });

    const dropdowns = getSheetDataAsJson(DROPDOWN_SHEET_NAME);
    const dropdownsMap = {};
    dropdowns.forEach((d, idx) => { dropdownsMap["item_" + idx] = d; });

    const data = {
      jobs: jobsMap,
      users: usersMap,
      dropdowns: dropdownsMap,
      customers: customersMap
    };
    
    const url = getFirebaseUrl('');
    const options = {
      method: "put",
      contentType: "application/json",
      payload: JSON.stringify(data),
      muteHttpExceptions: true
    };
    
    const res = UrlFetchApp.fetch(url, options);
    if (res.getResponseCode() === 200) {
      ui.alert('สำเร็จ', 'อัพโหลดข้อมูลไปยัง Firebase ในรูปแบบ JSON Map เรียบร้อยแล้ว!', ui.ButtonSet.OK);
    } else {
      throw new Error(res.getContentText());
    }
  } catch (error) {
    ui.alert('เกิดข้อผิดพลาด', 'ไม่สามารถอัพโหลดข้อมูลได้: ' + error.message, ui.ButtonSet.OK);
  }
}

function downloadDataFromFirebase() {
  const ui = SpreadsheetApp.getUi();
  const response = ui.alert('ยืนยันการดาวน์โหลด', 'คุณต้องการดาวน์โหลดข้อมูลจาก Firebase มาเขียนทับข้อมูลใน Google Sheets ใช่หรือไม่? (ข้อมูลปัจจุบันบน Google Sheets ทั้งหมดจะถูกเขียนทับ)', ui.ButtonSet.YES_NO);
  if (response !== ui.Button.YES) return;

  try {
    const url = getFirebaseUrl('');
    const res = UrlFetchApp.fetch(url, { method: "get", muteHttpExceptions: true });
    if (res.getResponseCode() !== 200) {
      throw new Error(res.getContentText());
    }
    
    const data = JSON.parse(res.getContentText());
    if (!data) {
      ui.alert('ไม่พบข้อมูล', 'ไม่มีข้อมูลใดๆ ใน Firebase Database', ui.ButtonSet.OK);
      return;
    }
    
    if (data.jobs !== undefined) {
      const jobsList = Object.values(data.jobs);
      writeJsonToSheet(JOB_DATA_SHEET_NAME, getJobHeaders(), jobsList);
    }
    if (data.users !== undefined) {
      const usersList = Object.values(data.users);
      writeJsonToSheet(USER_DATA_SHEET_NAME, getUserHeaders(), usersList);
    }
    if (data.dropdowns !== undefined) {
      const dropdownsList = Object.values(data.dropdowns);
      writeJsonToSheet(DROPDOWN_SHEET_NAME, getDropdownHeaders(), dropdownsList);
    }
    if (data.customers !== undefined) {
      const customersList = Object.values(data.customers);
      writeJsonToSheet(CUSTOMER_DATA_SHEET_NAME, getCustomerHeaders(), customersList);
    }
    
    ui.alert('สำเร็จ', 'ดาวน์โหลดข้อมูลจาก Firebase ลงสู่ Google Sheets เรียบร้อยแล้ว!', ui.ButtonSet.OK);
  } catch (error) {
    ui.alert('เกิดข้อผิดพลาด', 'ไม่สามารถดาวน์โหลดข้อมูลได้: ' + error.message, ui.ButtonSet.OK);
  }
}

function getSheetDataAsJson(sheetName) {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  const sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  
  const data = sheet.getRange(1, 1, lastRow, sheet.getLastColumn()).getValues();
  const headers = data.shift();
  
  return data.map(row => {
    const obj = {};
    headers.forEach((header, index) => {
      if (header) {
        let val = row[index];
        if (val instanceof Date) {
          obj[header] = val.toISOString();
        } else {
          obj[header] = val;
        }
      }
    });
    return obj;
  });
}

function writeJsonToSheet(sheetName, headers, jsonArray) {
  const ss = SpreadsheetApp.openById(SHEET_ID);
  let sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
  }
  
  sheet.clear();
  sheet.appendRow(headers);
  sheet.setFrozenRows(1);
  sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold");
  
  if (!jsonArray || jsonArray.length === 0) return;
  
  const rows = jsonArray.map(obj => {
    return headers.map(header => {
      let val = obj[header];
      if (val === undefined || val === null) return '';
      
      if (header.includes("เบอร์โทร") || header === "Phone" || header === "BillingPhone" || header === "ServicePhone") {
        val = String(val);
        if (!val.startsWith("'") && val !== "") {
          val = "'" + val;
        }
      }
      return val;
    });
  });
  
  sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
  
  if (sheetName === JOB_DATA_SHEET_NAME) sheet.getRange('C:C').setNumberFormat('@');
  if (sheetName === USER_DATA_SHEET_NAME) sheet.getRange('A:K').setNumberFormat('@');
  if (sheetName === CUSTOMER_DATA_SHEET_NAME) sheet.getRange('A:M').setNumberFormat('@');
}
